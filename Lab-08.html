fun1 <- function(n = 100, k = 4, lambda = 4) {
  x <- NULL
  
  for (i in 1:n){
    x <- rbind(x, rpois(k, lambda))    
  }
  
  return(x)
}

fun1alt <- function(n = 100, k = 4, lambda = 4) {
  matrix(rpois(n * k, lambda), nrow = n, ncol = k)
}

Problem 2: Parallelization

my_boot <- function(dat, stat, R, ncpus = 1L) {
  
  n <- nrow(dat)
  idx <- matrix(sample.int(n, n*R, TRUE), nrow=n, ncol=R)
  
  if (ncpus > 1L) {
    cl <- makeCluster(ncpus)
    on.exit(stopCluster(cl))
    clusterExport(cl, varlist = deparse(substitute(stat)), envir = parent.frame())
    ans <- parLapply(cl, seq_len(R), function(i) {
      stat(dat[idx[,i], , drop=FALSE])
    })
  } else {
    ans <- lapply(seq_len(R), function(i) {
      stat(dat[idx[,i], , drop=FALSE])
    })
  }
  
  ans <- do.call(rbind, ans)
  return(ans)
}

Problem 3. Short Answer

  1. A job is ideal for parallelization if we are performing the same computation many times; this allows us to run the jobs simultaneously and reduce the total run time of the code. Jobs that can be parallelized should not require communication with other results, they should be run independently.
  2. Vectorization refers to the process of running operations on a vector (which is faster in R) rather than running an operation one element at a time. This drastically cuts down on run time. Parallelization, as mentioned before, refers to running the same computationally intensive task in parallel across many cores to decrease run time. Vectorization should be done within a task while parallelization should be done if you want to perform that task many times.
