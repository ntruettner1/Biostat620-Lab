library(jsonlite)
url <- "https://github.com/datasciencelabs/2024/raw/refs/heads/main/data/regions.json"
regions <- fromJSON(url) # use fromJSON to read as a data.frame
regions <- regions %>%
  unnest(states)
regions <- regions %>%
  mutate(
    region_name = ifelse(region_name == "New York and New Jersey, Puerto Rico, Virgin Islands",
                         "NYPR",
                         region_name)
  )
regions <- regions %>%
  mutate(
    region = as.character(region)
  )
regions <- regions %>%
  mutate(region = factor(region))
regions <- regions %>%
  rename(state_name = states)
